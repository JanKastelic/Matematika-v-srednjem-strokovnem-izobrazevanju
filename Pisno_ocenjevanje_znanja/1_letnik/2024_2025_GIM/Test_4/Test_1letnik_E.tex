    \question[5]
    Natančno izračunajte vrednost izraza.
    $$ \dfrac{\sqrt{12a^5b^{-1}}}{2\sqrt{27ab^{-5}}}:\sqrt{a} $$

    \begin{solution}[7cm]
        \begin{align*} 
            \dfrac{\sqrt{5}}{\sqrt{5}+1}-\dfrac{5-3\sqrt{5}}{4\sqrt{5}}+\left(2+\sqrt{5}\right)^2-\sqrt{125}&=\dfrac{\sqrt{5}(\sqrt{5}-1)}{(\sqrt{5}+1)(\sqrt{5}-1)}-\dfrac{(5-3\sqrt{5})\sqrt{5}}{4\sqrt{5}\sqrt{5}}+4+4\sqrt{5}+5-5\sqrt{5}= \\
                                                                                                            &=\dfrac{5-\sqrt{5}}{5-1}-\dfrac{5\sqrt{5}-3\cdot 5}{4\cdot 5}+9-\sqrt{5}= \\
                                                                                                            &=\frac{5-\sqrt{5}}{4}-\dfrac{\sqrt{5}-3}{4}+9-\sqrt{5}= \\
                                                                                                            &=\frac{8-2\sqrt{5}}{4}+14-\sqrt{5}= \\
                                                                                                            &=11-\frac{3}{2}\sqrt{5}
        \end{align*}
        Za racionalizacijo $2$ točki, za izračun kvadrata dvočlenika $1$ točka, delno korenjenje $1$ točka, izračun $1$ točka, končne rezultat $1$ točka.
    \end{solution}

    \question[3]
    Rešite enačbo.
    $$ \dfrac{1}{8}\left(2+\dfrac{2}{3}x\right)-\dfrac{1}{4}\left(\dfrac{x}{2}-\dfrac{1}{6}\right)=0 $$
    
    
    \begin{solution}[4cm]
        \begin{align*} 
            \dfrac{2}{x-3}-\dfrac{x}{x-1} &=\dfrac{2x}{x^2-4x+3} \\
            2(x-1)-x(x-3) &= 2x \\
            2x-2-x^2+3x-2x &=0 \\
            x^2-3x+2 &=0 \\
            (x-1)(x-2) &=0
        \end{align*}
        Pravilno odpravljanje ulomkov oziroma skupni imenovalec $1$ točka, zapis in upoštevanje pogojev za rešitev ($x\notin\{1,3\}$) $1$ točka, pravilno reševanje razcepne enačbe $1$ točka, zapis rešitve $x=2$ $1$ točka.
    \end{solution}
